# Function to calculate binomial coefficient using Dynamic Programming
def binomial_coefficient(n, r):

    # Create a DP table of size (n+1) x (r+1)
    dp = [[0 for j in range(r + 1)] for i in range(n + 1)]

    # Fill the DP table
    for i in range(n + 1):

        for j in range(min(i, r) + 1):

            # Base case: nC0 = 1 and nCn = 1
            if j == 0 or j == i:
                dp[i][j] = 1

            else:
                # Pascal's Triangle formula
                dp[i][j] = dp[i - 1][j - 1] + dp[i - 1][j]

    # Return the required binomial coefficient
    return dp[n][r]


# MAIN PROGRAM

# Accept the value of n
n = int(input("Enter the value of n: "))

# Accept the value of r
r = int(input("Enter the value of r: "))

# Check whether the input is valid
if n < 0 or r < 0 or r > n:
    print("Invalid input! Make sure n >= r >= 0.")

else:
    # Calculate the binomial coefficient
    result = binomial_coefficient(n, r)

    # Display the result
    print("Binomial Coefficient C(", n, ",", r, ") =", result)
